# select partcular sheets
import pandas as pd
from sqlalchemy import create_engine

excel_file = r"your excel path"

server = your server name
database = "your database name"

connection_string = (
    f"mssql+pyodbc://{server}/{database}"
    "?driver=ODBC+Driver+17+for+SQL+Server"
    "&trusted_connection=yes"
)

engine = create_engine(connection_string)

# Tables to import
sheets = [
    "FactOrderLine",
    "DimProduct",
    "DimCustomer"
]

for sheet_name in sheets:

    df = pd.read_excel(
        excel_file,
        sheet_name=sheet_name
    )

    df.to_sql(
        name=sheet_name,
        con=engine,
        schema="dbo",
        if_exists="replace",
        index=False
    )

    print(f"Imported: {sheet_name} -> dbo.{sheet_name}")

engine.dispose()

print("Selected tables imported successfully!")
